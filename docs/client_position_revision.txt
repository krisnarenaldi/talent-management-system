3. Service Layer – Bulk Reject Logic
Buat file baru backend/app/services/application_service.py

from sqlalchemy.orm import Session
from sqlalchemy import update, func
from app.models.application import Application

def bulk_reject_applications(
	db: Session,
	position_ids: list[str],
	reason: str,
) -> int:
"""
Semua Application dengan status='active' pada posisi‑posisi yang diberikan
di‑set menjadi 'rejected' + alasan.
Mengembalikan jumlah baris yang ter‑update.
"""
stmt = (
	update(Application)
	.where(
		Application.position_id.in_(position_ids),
		Application.status == "active",
	)
	.values(
		status="rejected",
		reject_reason=reason,
		updated_at=func.now(),
	)
)
result = db.execute(stmt)
return result.rowcount

4. Extend API Endpoints
4.1 PUT /positions/{position_id}

Edit backend/app/routers/positions.py:

from app.services.application_service import bulk_reject_applications

@router.put("/{position_id}", response_model=PositionResponse)
def update_position(
	position_id: str,
	payload: PositionUpdate,
	db: Session = Depends(get_db),
	current_user=Depends(require_role("admin", "manager")),
):
position = db.query(Position).filter(Position.id == position_id).first()
if not position:
raise HTTPException(status_code=404, detail="Position tidak ditemukan")

# ---------- Bulk reject flow ----------
if payload.is_active is False and payload.bulk_reject:
if not payload.reject_reason:
raise HTTPException(
	status_code=400,
	detail="reject_reason wajib bila bulk_reject=True",
)
# Mulai transaksi atomik
with db.begin():
position.is_active = False
# reject semua lamaran aktif pada posisi ini
bulk_reject_applications(db, [position.id], payload.reject_reason)

# ---------- Normal updates ----------
for key, value in payload.model_dump(exclude_unset=True).items():
setattr(position, key, value)

db.commit()
db.refresh(position)
return position

6. Front‑end Adjustments (Next.js – App Router)
6.1 Komponen tombol “Non‑aktifkan Posisi”
// src/app/(dashboard)/admin/positions/[clientId]/page.tsx
import { useRouter } from "next/navigation";

export default function DeactivatePositionButton({ position }: { position: Position }) {
	const router = useRouter();
	
	const handleDeactivate = async () => {
		if (!confirm(
			"Posisi akan dinon‑aktifkan. Semua kandidat yang masih aktif akan otomatis ditolak. Lanjutkan?"
		)) return;
		
		const res = await fetch(`/api/positions/${position.id}`, {
			method: "PUT",
			headers: { "Content-Type": "application/json" },
			body: JSON.stringify({
				is_active: false,
				bulk_reject: true,
				reject_reason: "Posisi ditutup oleh HR",
			}),
		});
		
		if (!res.ok) {
			const err = await res.json();
			alert(`Gagal: ${err.detail}`);
			return;
		}
		router.refresh();        // reload list / detail
	};
	
	return (
		<Button color="danger" onClick={handleDeactivate}>
		Non‑aktifkan Posisi
		</Button>
	);
}
